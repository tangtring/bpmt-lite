> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `v1.6.2` as a patch release that fixes GitHub issues `#12` and `#13`, makes one-line install default to the full database, adds a non-invasive upgrade script, and rewrites README as a clean entry document.

**Architecture:** Keep product behavior in `bpmt-web/platform`; do not change `bpmt-api` contracts. Treat install and upgrade as file-system deployment operations: install prepares a runnable directory, while upgrade pulls BPMT-owned `latest` images, downloads version reference files, runs version SQL scripts, and records state under `.bpmt-lite/` without writing deployment metadata into MariaDB.

**Tech Stack:** Java 8, Maven 3, Tomcat 7, JSP, JUnit 4, Spring mock web tests, POSIX shell, Docker Compose, MariaDB 10.11.

- [ ] **Step 2: Create `scripts/upgrade.sh`**

Create `scripts/upgrade.sh` with POSIX shell. Required functions:

```sh
#!/bin/sh
set -eu

REPO="${BPMT_REPO:-wodenwang/bpmt-lite}"
GITHUB_API="${BPMT_GITHUB_API:-https://api.github.com}"
RAW_BASE_OVERRIDE="${BPMT_RAW_BASE_URL:-}"
STATE_DIR="${BPMT_UPGRADE_STATE_DIR:-.bpmt-lite}"
APPLIED_LOG="$STATE_DIR/applied-upgrades.log"
LAST_VERSION_FILE="$STATE_DIR/last-version"
DRY_RUN=0
FROM_VERSION=""
TARGET_VERSION=""

usage() {
  echo "Usage: upgrade.sh [--dry-run] [--from vX.Y.Z] [vX.Y.Z]" >&2
}

normalize_version() {
  value="$1"
  case "$value" in
    v*) echo "$value" ;;
    "") echo "" ;;
    *) echo "v$value" ;;
  esac
}

download() {
  url="$1"
  target="$2"
  tmp="$target.tmp"
  curl -fsSL "$url" -o "$tmp"
  mv "$tmp" "$target"
}

latest_release() {
  if command -v curl >/dev/null 2>&1; then
    tag="$(curl -fsSL "$GITHUB_API/repos/$REPO/releases/latest" 2>/dev/null \
      | sed -n 's/.*"tag_name"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p' \
      | head -n 1)"
    if [ -n "$tag" ]; then
      normalize_version "$tag"
      return 0
    fi
  fi
  if command -v git >/dev/null 2>&1; then
    git ls-remote --tags "https://github.com/$REPO.git" 'refs/tags/v*' \
      | sed 's#.*refs/tags/##; s#\\^{}##' \
      | sort -V \
      | tail -n 1
    return 0
  fi
  return 1
}

raw_base_for() {
  version="$1"
  if [ -n "$RAW_BASE_OVERRIDE" ]; then
    echo "$RAW_BASE_OVERRIDE"
  else
    echo "https://raw.githubusercontent.com/$REPO/$version"
  fi
}
```

Argument parsing must accept:

```sh
while [ "$#" -gt 0 ]; do
  case "$1" in
    --dry-run)
      DRY_RUN=1
      ;;
    --from)
      shift
      FROM_VERSION="$(normalize_version "${1:-}")"
      ;;
    -h|--help|help)
      usage
      exit 0
      ;;
    v*|[0-9]*)
      TARGET_VERSION="$(normalize_version "$1")"
      ;;
    *)
      usage
      exit 2
      ;;
  esac
  shift
done
```

Implement pull and SQL execution:

```sh

sql_key="$(sha256sum "$UPGRADE_SQL" 2>/dev/null | awk '{print $1}')"
if grep -F "$TARGET_VERSION $UPGRADE_SQL $sql_key" "$APPLIED_LOG" >/dev/null 2>&1; then
  echo "Upgrade SQL already recorded: $UPGRADE_SQL"
else
  docker compose exec -T bpmt-mariadb mariadb -uroot -p"${DB_PASSWORD:-123456}" < "$UPGRADE_SQL"
  printf '%s %s %s %s\n' "$(date '+%Y-%m-%dT%H:%M:%S%z')" "$TARGET_VERSION" "$UPGRADE_SQL" "$sql_key" >> "$APPLIED_LOG"
fi

echo "bpmt-lite upgraded to $TARGET_VERSION using latest BPMT images."
```

```sh
hash_file() {

    sha256sum "$1" | awk '{print $1}'
  else
    shasum -a 256 "$1" | awk '{print $1}'
  fi
}
```
