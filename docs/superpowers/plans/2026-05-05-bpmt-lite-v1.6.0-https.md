# bpmt-lite v1.6.0 HTTPS Support Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add HTTPS support for the `bpmt-lite` nginx single entry while keeping Web, UEditor, OAuth, H5, and API working under both built-in TLS and trusted upstream TLS.

**Architecture:** Keep TLS termination at `bpmt-nginx`; keep `bpmt-web` and `bpmt-api` on internal HTTP. Generate nginx config from tracked templates and environment variables. Make Java URL generation proxy-aware in `Actions.Util`, then clean only runtime-blocking frontend HTTP resources and extend smoke checks.

**Tech Stack:** Java 8, JUnit 4, Spring test mocks, Maven with `settings.local.xml`, shell scripts, Docker Compose, nginx 1.27-alpine, Playwright/manual browser smoke.

## Scope Check

The spec covers one release theme: HTTPS runtime correctness for the existing single-entry deployment. It touches nginx, Java URL generation, runtime JSP resources, smoke scripts, and docs because those pieces must work together before HTTPS can be accepted. The work should stay in one plan and should not expand into ACME automation, Tomcat TLS, HSTS-by-default, frontend framework rewrites, or full-text removal of every `http://` string.

```java
public static String getContextPath(HttpServletRequest request) {
    return getPublicBaseUrl(request) + request.getContextPath();
}

public static String getFullURL(HttpServletRequest request) {
    String url = RequestUtils.getStringValue(request, Keys.FULL_URL.toString());
    if (StringUtils.isNotEmpty(url)) {
        return url;
    }

    StringBuilder publicUrl = new StringBuilder(getPublicBaseUrl(request));
    publicUrl.append(request.getRequestURI());
    String queryString = request.getQueryString();
    if (StringUtils.isNotEmpty(queryString)) {
        publicUrl.append('?').append(queryString);
    }
    url = publicUrl.toString();

    if (logger.isDebugEnabled()) {
        logger.debug("当前FULL URL:{}", url);
    }
    return url;
}

private static String getPublicBaseUrl(HttpServletRequest request) {
    String scheme = firstHeaderValue(request, "X-Forwarded-Proto");
    if (StringUtils.isBlank(scheme)) {
        scheme = request.getScheme();
    }
    scheme = scheme.toLowerCase();

    String host = firstHeaderValue(request, "X-Forwarded-Host");
    if (StringUtils.isBlank(host)) {
        host = firstHeaderValue(request, "Host");
    }

    String port = null;
    if (StringUtils.isBlank(host)) {
        host = request.getServerName();
        port = String.valueOf(request.getServerPort());
    } else {
        HostAndPort hostAndPort = splitHostAndPort(host);
        host = hostAndPort.host;
        port = hostAndPort.port;
    }

    if (StringUtils.isBlank(port)) {
        port = firstHeaderValue(request, "X-Forwarded-Port");
    }
    if (StringUtils.isBlank(port)) {
        port = String.valueOf(request.getServerPort());
    }

    StringBuilder base = new StringBuilder();
    base.append(scheme).append("://").append(host);
    if (!isDefaultPort(scheme, port)) {
        base.append(':').append(port);
    }
    return base.toString();
}

private static String firstHeaderValue(HttpServletRequest request, String name) {
    String value = request.getHeader(name);
    if (StringUtils.isBlank(value)) {
        return null;
    }
    int comma = value.indexOf(',');
    if (comma >= 0) {
        value = value.substring(0, comma);
    }
    value = value.trim();
    return value.length() == 0 ? null : value;
}

private static boolean isDefaultPort(String scheme, String port) {
    return ("http".equalsIgnoreCase(scheme) && "80".equals(port))
            || ("https".equalsIgnoreCase(scheme) && "443".equals(port));
}

private static HostAndPort splitHostAndPort(String hostHeader) {
    String host = hostHeader == null ? "" : hostHeader.trim();
    String port = null;
    if (host.startsWith("[")) {
        int end = host.indexOf(']');
        if (end >= 0 && host.length() > end + 2 && host.charAt(end + 1) == ':') {
            port = host.substring(end + 2);
            host = host.substring(0, end + 1);
        }
    } else {
        int colon = host.lastIndexOf(':');
        if (colon > 0 && host.indexOf(':') == colon) {
            port = host.substring(colon + 1);
            host = host.substring(0, colon);
        }
    }
    return new HostAndPort(host, port);
}

private static class HostAndPort {
    private final String host;
    private final String port;

    private HostAndPort(String host, String port) {
        this.host = host;
        this.port = port;
    }
}
```

```sh
#!/bin/sh
set -eu

CERT_DIR="${BPMT_TLS_CERT_DIR:-certs}"
CERT_FILE="${BPMT_TLS_CERT_FILE_HOST:-$CERT_DIR/fullchain.pem}"
KEY_FILE="${BPMT_TLS_KEY_FILE_HOST:-$CERT_DIR/privkey.pem}"
HOSTS="${BPMT_TLS_HOSTS:-localhost}"
IPS="${BPMT_TLS_IPS:-127.0.0.1}"
DAYS="${BPMT_TLS_DAYS:-365}"
FORCE="${BPMT_TLS_FORCE:-false}"

if ! command -v openssl >/dev/null 2>&1; then
  echo "openssl is required to generate a self-signed certificate." >&2
  exit 1
fi

if [ -f "$CERT_FILE" ] && [ -f "$KEY_FILE" ] && [ "$FORCE" != "1" ] && [ "$FORCE" != "true" ]; then
  echo "TLS certificate already exists: $CERT_FILE"
  echo "TLS key already exists: $KEY_FILE"
  exit 0
fi

mkdir -p "$CERT_DIR"
TMP_CONF="$CERT_DIR/openssl-san.cnf.tmp"
SAN=""

OLD_IFS="$IFS"
IFS=","
for host in $HOSTS; do
  host="$(printf '%s' "$host" | sed 's/^ *//;s/ *$//')"
  if [ -n "$host" ]; then
    SAN="${SAN}DNS:${host},"
  fi
done
for ip in $IPS; do
  ip="$(printf '%s' "$ip" | sed 's/^ *//;s/ *$//')"
  if [ -n "$ip" ]; then
    SAN="${SAN}IP:${ip},"
  fi
done
IFS="$OLD_IFS"
SAN="$(printf '%s' "$SAN" | sed 's/,$//')"

cat > "$TMP_CONF" <<EOF
[req]
distinguished_name=req_distinguished_name
x509_extensions=v3_req
prompt=no
[req_distinguished_name]
CN=bpmt-lite.local
[v3_req]
subjectAltName=$SAN
EOF

openssl req -x509 -nodes -newkey rsa:2048 -days "$DAYS" \
  -keyout "$KEY_FILE" \
  -out "$CERT_FILE" \
  -config "$TMP_CONF" >/dev/null 2>&1
rm -f "$TMP_CONF"
chmod 600 "$KEY_FILE"

echo "Generated self-signed TLS certificate: $CERT_FILE"
echo "Generated self-signed TLS key: $KEY_FILE"
echo "Browsers will warn until this certificate is trusted locally."
```

Create `scripts/render-nginx-conf.sh`:

```sh
#!/bin/sh
set -eu

TEMPLATE="${BPMT_NGINX_TEMPLATE:-docker/nginx/nginx.conf.template}"
OUTPUT="${BPMT_NGINX_CONF:-docker/nginx/nginx.conf}"
HTTPS_ENABLED="${BPMT_HTTPS_ENABLED:-false}"
HTTP_REDIRECT="${BPMT_HTTP_REDIRECT:-true}"
HTTPS_PORT="${BPMT_HTTPS_PORT:-443}"
CERT_FILE="${BPMT_TLS_CERT_FILE:-/etc/nginx/certs/fullchain.pem}"
KEY_FILE="${BPMT_TLS_KEY_FILE:-/etc/nginx/certs/privkey.pem}"

mkdir -p "$(dirname "$OUTPUT")"

write_proxy_locations() {
  cat <<'EOF'
    client_max_body_size 100m;

    location /api/ {
        proxy_pass http://bpmt-api:8081/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $http_host;
        proxy_set_header X-Forwarded-Host $http_host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Real-IP $remote_addr;
    }

    location /ueditor/ {
        proxy_pass http://bpmt-web:8080/ueditor/;
        proxy_http_version 1.1;
        proxy_set_header Host $http_host;
        proxy_set_header X-Forwarded-Host $http_host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Real-IP $remote_addr;
    }

    location / {
        proxy_pass http://bpmt-web:8080/;
        proxy_http_version 1.1;
        proxy_set_header Host $http_host;
        proxy_set_header X-Forwarded-Host $http_host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Real-IP $remote_addr;
    }
EOF
}

case "$HTTPS_ENABLED" in
  1|true|TRUE|yes|YES)
    if [ "$HTTP_REDIRECT" = "0" ] || [ "$HTTP_REDIRECT" = "false" ]; then
      {
        printf '%s\n' 'server {'
        printf '%s\n' '    listen 80;'
        printf '%s\n' '    server_name _;'
        write_proxy_locations
        printf '%s\n\n' '}'
      } > "$OUTPUT"
    else
      {
        printf '%s\n' 'server {'
        printf '%s\n' '    listen 80;'
        printf '%s\n' '    server_name _;'
        printf '%s\n' '    return 301 https://$http_host$request_uri;'
        printf '%s\n\n' '}'
      } > "$OUTPUT"
    fi
    {
      printf '%s\n' 'server {'
      printf '%s\n' '    listen 443 ssl;'
      printf '%s\n' '    server_name _;'
      printf '    ssl_certificate %s;\n' "$CERT_FILE"
      printf '    ssl_certificate_key %s;\n' "$KEY_FILE"
      write_proxy_locations
      printf '%s\n' '}'
    } >> "$OUTPUT"
    ;;
  *)
    {
      printf '%s\n' 'server {'
      printf '%s\n' '    listen 80;'
      printf '%s\n' '    server_name _;'
      write_proxy_locations
      printf '%s\n' '}'
    } > "$OUTPUT"
    ;;
esac

echo "Rendered nginx config: $OUTPUT"
```
