                Class<?> returnType = method.getReturnType();
                if (Boolean.TYPE.equals(returnType)) {
                    return Boolean.FALSE;

                if (Integer.TYPE.equals(returnType)) {
                    return Integer.valueOf(0);
                }
                return null;
            }

public void form(HttpServletRequest request, HttpServletResponse response) {
    FlowObject fo = FlowObject.create(request);
    String taskId;
    String pdId;
    String pdKey;
    String ordId;

    if (StringUtils.isNotEmpty(fo.getTaskId())) {// 任务入参
        Task task = findTaskById(fo.getTaskId());
        if (task == null) {
            throw new SystemRuntimeException(ExceptionType.FLOW_TASK_ASSIGNEE, "任务不存在,可能已被处理.");
        }
        ProcessInstance pi = findProcessInstanceById(task.getProcessInstanceId());
        if (pi == null) {
            throw new SystemRuntimeException(ExceptionType.BUSINESS, "任务[" + fo.getTaskId() + "]对应的流程实例不存在.");
        }
        pdId = pi.getProcessDefinitionId();
        taskId = task.getId();
        pdKey = "";
        ordId = resolveTaskOrdId(task, pi);
    } else if (StringUtils.isNotEmpty(fo.getOrdId()) && StringUtils.isNotEmpty(fo.getPdId())) {// 订单号+流程ID入参
        pdId = fo.getPdId();
        ordId = fo.getOrdId();
        List<Task> tasks = FlowFactory.getTaskService().createTaskQuery().processDefinitionId(pdId)
                .processInstanceBusinessKey(fo.getOrdId()).orderByTaskCreateTime().asc().list();
        taskId = null;
        if (tasks.size() > 0) {
            for (Task o : tasks) {
                if (FlowHelper.checkTask(o)) {
                    taskId = o.getId();
                    break;
                }
            }
            if (taskId == null) {
                throw new SystemRuntimeException(ExceptionType.BUSINESS, "无权处理订单.");
            }
        } else {
            taskId = "";
        }
        pdKey = "";
    } else if (StringUtils.isNotEmpty(fo.getPdKey())) {
        taskId = "";
        ProcessDefinition pd = FlowService.getInstance().getPd(fo.getPdKey(), 0);
        if (pd == null) {
            throw new SystemRuntimeException(ExceptionType.BUSINESS, "入参错误,找不到流程[" + fo.getPdKey() + "].");
        }
        pdKey = fo.getPdKey();
        pdId = pd.getId();
        ordId = fo.getOrdId();
    } else {
        throw new SystemRuntimeException(ExceptionType.BUSINESS, "入参错误,无法查看订单.");
    }

    WfPd config = findPdConfigById(pdId);
    VwUrl basicUrl = findBasicUrl(config);
    if (config == null || basicUrl == null) {
        throw new SystemRuntimeException(ExceptionType.BUSINESS, "流程已被删除,无法浏览订单.");
    }
    String params = "{form:true,pdKey:'" + pdKey + "'}";
    redirectAction(request, response,
            basicUrl.getUrl() + "?" + Keys.PARAMS.toString() + "=" + params + "&"
                    + FlowKeys._TASK_ID.getName() + "=" + taskId + "&"
                    + FlowKeys._ORD_ID.getName() + "=" + requireRedirectOrdId(taskId, ordId));
}

protected Task findTaskById(String taskId) {
    return FlowFactory.getTaskService().createTaskQuery().taskId(taskId).singleResult();

protected VwUrl findBasicUrl(WfPd config) {
    if (config == null) {
        return null;
    }
    return (VwUrl) ORMService.getInstance().findByPk(VwUrl.class.getName(), config.getBasicViewKey());

protected Map<String, Object> findHistoryOrderByTask(String tableName, String taskId) {
    return (Map<String, Object>) ORMAdapterService.getInstance().find(tableName,
            new DataCondition().setStringEqual(OrderHistoryModelKeys.TASK_ID.name(), taskId).toEntity());
}

protected void redirectAction(HttpServletRequest request, HttpServletResponse response, String actionUrl) {
    Actions.redirectAction(request, response, actionUrl);
}

private String resolveTaskOrdId(Task task, ProcessInstance pi) {
    if (pi != null && StringUtils.isNotBlank(pi.getBusinessKey())) {
        return pi.getBusinessKey();
    }
    String taskId = task == null ? null : task.getId();
    String historyTableName = null;
    if (pi != null && pi.getProcessVariables() != null) {
        Object value = pi.getProcessVariables().get(VariableKeys._ORDER_HISTORY_TABLE_NAME.name());
        if (value != null) {
            historyTableName = value.toString();
        }
    }
    if (StringUtils.isNotBlank(historyTableName) && StringUtils.isNotBlank(taskId)) {
        Map<String, Object> historyOrder = findHistoryOrderByTask(historyTableName, taskId);
        if (historyOrder != null) {
            Object ordId = historyOrder.get(OrderHistoryModelKeys.ORD_ID.name());
            if (ordId != null && StringUtils.isNotBlank(ordId.toString())) {
                return ordId.toString();
            }
        }
    }
    throw new SystemRuntimeException(ExceptionType.BUSINESS, "无法解析任务[" + taskId + "]对应的订单号.");
}

private String requireRedirectOrdId(String taskId, String ordId) {
    if (StringUtils.isBlank(ordId)) {
        throw new SystemRuntimeException(ExceptionType.BUSINESS, "无法解析任务[" + taskId + "]对应的订单号.");
    }
    return ordId;
}
