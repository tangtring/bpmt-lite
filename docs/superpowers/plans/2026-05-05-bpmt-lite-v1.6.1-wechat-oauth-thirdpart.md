# v1.6.1 WeChat OAuth Thirdparty Handoff Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build v1.6.1 so third-party OAuth clients can bind to BPMT WeChat login and receive the normal OAuth authorization-code callback after WeChat login creates the BPMT session.

**Architecture:** Keep `/oauth/authorize` as the only third-party entrypoint. Extend `CM_THIRDPART` with explicit WeChat login binding fields, add a focused `OAuthWechatLoginService` plus provider boundary, and call it only when authorize has no BPMT login, the request is from WeChat, and the client is configured for WeChat login. Production uses existing WeChat SDK calls; tests and local smoke use an explicit fake provider.

**Tech Stack:** Java 8, Maven 3, Tomcat 7, Spring mock test utilities, JUnit 4, Hibernate HBM, MariaDB SQL, JSP, Docker Compose.

---

## Scope Check

This is one cohesive subsystem: third-party OAuth authorize login-state handoff inside `bpmt-web/platform`. The work touches data model, service boundary, authorize integration, admin configuration UI, docs, and release verification because those are coupled by the new `CM_THIRDPART` fields. It does not require splitting into separate plans.

```java
public static Map<String, Object> normalizeWechatLogin(Map<String, Object> input) {
    Map<String, Object> normalized = new HashMap<String, Object>();
    Integer enabled = normalizeActiveFlag(input == null ? null : input.get("wechatLoginEnabled"));
    normalized.put("wechatLoginEnabled", enabled);
    if (!Integer.valueOf(1).equals(enabled)) {
        normalized.put("wechatType", null);
        normalized.put("wechatKey", null);
        normalized.put("wechatScope", null);
        return normalized;
    }

    String wechatType = trimToNull(input.get("wechatType"));
    String wechatKey = trimToNull(input.get("wechatKey"));
    String wechatScope = trimToNull(input.get("wechatScope"));
    if (!WECHAT_TYPE_AGENT.equals(wechatType) && !WECHAT_TYPE_MP.equals(wechatType)) {
        throw new SystemRuntimeException(ExceptionType.BUSINESS, "wechatType格式不正确.");
    }
    if (StringUtils.isBlank(wechatKey) || !SAFE_KEY.matcher(wechatKey).matches()) {
        throw new SystemRuntimeException(ExceptionType.BUSINESS, "wechatKey格式不正确.");
    }

    normalized.put("wechatLoginEnabled", Integer.valueOf(1));
    normalized.put("wechatType", wechatType);
    normalized.put("wechatKey", wechatKey);
    if (WECHAT_TYPE_MP.equals(wechatType)) {
        if (StringUtils.isBlank(wechatScope)) {
            wechatScope = WECHAT_SCOPE_BASE;
        }
        if (!WECHAT_SCOPE_BASE.equals(wechatScope) && !WECHAT_SCOPE_USERINFO.equals(wechatScope)) {
            throw new SystemRuntimeException(ExceptionType.BUSINESS, "wechatScope格式不正确.");
        }
        normalized.put("wechatScope", wechatScope);
    } else {
        normalized.put("wechatScope", null);
    }
    return normalized;
}

private static void applyWechatLogin(DataPO po, Map<String, Object> input) {
    Map<String, Object> wechat = normalizeWechatLogin(input);
    po.set("wechatLoginEnabled", wechat.get("wechatLoginEnabled"));
    po.set("wechatType", wechat.get("wechatType"));
    po.set("wechatKey", wechat.get("wechatKey"));
    po.set("wechatScope", wechat.get("wechatScope"));
}
```

```java
private String readResource(String name) throws Exception {
    InputStream input = resource(name);
    org.junit.Assert.assertNotNull(input);
    try {
        java.io.ByteArrayOutputStream out = new java.io.ByteArrayOutputStream();
        byte[] buffer = new byte[4096];
        int read;
        while ((read = input.read(buffer)) >= 0) {

        input.close();
    }
```

```java
package com.riversoft.module.oauth.wechat;

import java.util.Map;

import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

import org.apache.commons.lang3.StringUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.riversoft.core.web.Actions;
import com.riversoft.module.thirdpart.ThirdpartService;

public class OAuthWechatLoginService {
    private static final Logger logger = LoggerFactory.getLogger(OAuthWechatLoginService.class);

    private final WechatOAuthProvider provider;

    public OAuthWechatLoginService() {
        this(new RealWechatOAuthProvider());
    }

    public OAuthWechatLoginService(WechatOAuthProvider provider) {
        this.provider = provider;
    }

    public OAuthWechatLoginResult handle(HttpServletRequest request, HttpServletResponse response,
            Map<String, Object> thirdpart) {
        if (!Actions.Util.fromWx(request) || thirdpart == null) {
            return OAuthWechatLoginResult.skip();
        }
        if (!Integer.valueOf(1).equals(thirdpart.get("wechatLoginEnabled"))) {
            return OAuthWechatLoginResult.skip();
        }

        String clientId = stringValue(thirdpart.get("clientId"));
        String thirdpartKey = stringValue(thirdpart.get("thirdpartKey"));
        Map<String, Object> normalized;
        try {
            normalized = ThirdpartService.normalizeWechatLogin(thirdpart);
        } catch (RuntimeException e) {
            logger.info("OAuth WeChat login. clientId={} thirdpartKey={} result={} reason={}", clientId, thirdpartKey,
                    "deny", "wechat_config_invalid");
            return OAuthWechatLoginResult.error("wechat_config_invalid", "外部系统微信登录配置不正确。");
        }

        String wechatType = stringValue(normalized.get("wechatType"));
        String wechatKey = stringValue(normalized.get("wechatKey"));
        String wechatScope = stringValue(normalized.get("wechatScope"));
        String code = request.getParameter("code");
        if (StringUtils.isBlank(code)) {
            String callbackUrl = Actions.Util.getFullURL(request);
            String redirectUrl = provider.buildAuthorizationUrl(wechatType, wechatKey, wechatScope, callbackUrl);
            logger.info("OAuth WeChat login. clientId={} thirdpartKey={} wechatType={} wechatKey={} result={} reason={}",
                    clientId, thirdpartKey, wechatType, wechatKey, "pending", "wechat_redirect_issued");
            return OAuthWechatLoginResult.redirect(redirectUrl);
        }

        try {
            String userId = provider.loginByCode(request, wechatType, wechatKey, wechatScope, code);
            logger.info("OAuth WeChat login. clientId={} thirdpartKey={} wechatType={} wechatKey={} userId={} result={} reason={}",
                    clientId, thirdpartKey, wechatType, wechatKey, userId, "allow", "wechat_login_success");
            return OAuthWechatLoginResult.loggedIn(userId);
        } catch (RuntimeException e) {
            logger.info("OAuth WeChat login. clientId={} thirdpartKey={} wechatType={} wechatKey={} result={} reason={}",
                    clientId, thirdpartKey, wechatType, wechatKey, "deny", "wechat_login_failed");
            return OAuthWechatLoginResult.error("wechat_login_failed", "微信登录失败，请联系系统管理员检查微信配置。");
        }
    }

    private String stringValue(Object value) {
        return value == null ? null : String.valueOf(value);
    }
}
```

````markdown
# bpmt-lite v1.6.1 发布记录

## 版本定位

`v1.6.1` 基于 `v1.6.0`，增强微信生态下第三方 OAuth 登录态传导。

## 主要变化

- `CM_THIRDPART` 增加微信登录绑定字段。
- `/oauth/authorize` 在微信 UA、无 BPMT 登录态、外部系统启用微信登录绑定时，会先进入企业号或服务号微信 OAuth 登录。
- 微信登录成功后继续标准 OAuth Authorization Code 回调。
- 本机提供 fake provider 验证链路，真实微信作为部署后人工验收项。

scripts/build-image.sh
scripts/build-api-image.sh
scripts/smoke-oauth-wechat.sh
```

## 发布边界

- 不新增第三方通知 API。
- 不实现 OIDC、refresh token 或单点登出。
- 不改变 `bpmt-api` 对外业务接口。
````
